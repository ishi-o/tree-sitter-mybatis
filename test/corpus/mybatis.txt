    SELECT * FROM users WHERE id = #id#
    <isNotEmpty prepend="AND" property="status">
      status = #status#
    </isNotEmpty>
  </select>
  <sql id="baseColumns">id, name</sql>
  <include refid="baseColumns"/>
</sqlMap>
